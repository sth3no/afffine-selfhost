# pyproject.toml keeps loose `>=` ranges (what the code needs);
# this file records exact versions of a combination that passed the
# full test suite. The Dockerfile and CI install with
# `pip install -c constraints.txt`, so a Portainer rebuild resolves
# the same versions every time instead of whatever shipped upstream
# that morning.
#
# Deliberately NOT pinned: yt-dlp — its value is tracking the platform
# extractor arms race, so it always resolves latest at build time.
#

#   python3.12 -m venv .venv && .venv/bin/pip install -e ".[dev]"
#   .venv/bin/python -m pytest tests/            # must be green
#   .venv/bin/pip freeze --exclude-editable | grep -vi '^yt-dlp==' > constraints.txt
